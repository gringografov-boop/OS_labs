cmake_minimum_required(VERSION 3.16)
project(os_lab2_variant5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(common STATIC inc/common.cpp)
target_include_directories(common PUBLIC inc/include)
add_executable(parent inc/parent.cpp)
target_link_libraries(parent PRIVATE common)
add_executable(child inc/child.cpp)
target_link_libraries(child PRIVATE common)
set_target_properties(parent child PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(MSVC)
  target_compile_options(common PRIVATE /W4 /permissive-)
  target_compile_options(parent PRIVATE /W4 /permissive-)
  target_compile_options(child PRIVATE /W4 /permissive-)
else()
  target_compile_options(common PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(parent PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(child PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
