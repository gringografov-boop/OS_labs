cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(lab4_variant5 LANGUAGES CXX)
endif()
if(NOT TARGET os)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../common/code/os" "${CMAKE_CURRENT_BINARY_DIR}/os")
endif()
# ParseInteger / IsPrime / UniqueFile are shared with lab 2.
if(NOT TARGET lab2_logic)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../lab2" "${CMAKE_CURRENT_BINARY_DIR}/lab2")
endif()

add_library(lab4_ipc STATIC)
target_compile_features(lab4_ipc PUBLIC cxx_std_17)
target_include_directories(lab4_ipc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/code/include")
if(WIN32)
    target_sources(lab4_ipc PRIVATE code/ipc_win.cpp)
    target_compile_definitions(lab4_ipc PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
else()
    target_sources(lab4_ipc PRIVATE code/ipc_posix.cpp)
    find_package(Threads REQUIRED)
    target_link_libraries(lab4_ipc PUBLIC Threads::Threads)
endif()

add_executable(lab4_parent code/parent/main.cpp)
add_executable(lab4_child code/child/main.cpp)
target_link_libraries(lab4_parent PRIVATE os lab4_ipc)
target_link_libraries(lab4_child PRIVATE os lab4_ipc lab2_logic)
if(WIN32)
    target_compile_definitions(lab4_child PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
foreach(target lab4_parent lab4_child)
    set_target_properties(${target} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/lab4")
    foreach(config Debug Release RelWithDebInfo MinSizeRel)
        string(TOUPPER "${config}" upper)
        set_target_properties(${target} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin/lab4")
    endforeach()
endforeach()
add_dependencies(lab4_parent lab4_child)
if(MINGW)
    target_link_options(lab4_child PRIVATE -municode)
endif()
foreach(target lab4_parent lab4_child lab4_ipc)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
