cmake_minimum_required(VERSION 3.16)
project(os LANGUAGES CXX)
add_library(os STATIC os.cpp)
if(WIN32)
    target_sources(os PRIVATE os_win.cpp)
    target_compile_definitions(os PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(os PRIVATE os_posix.cpp)
    target_compile_definitions(os PRIVATE _GNU_SOURCE)
else()
    message(FATAL_ERROR "Supported systems: Linux and Windows")
endif()
target_compile_features(os PUBLIC cxx_std_17)
target_include_directories(os PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(MSVC)
    target_compile_options(os PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(os PRIVATE -Wall -Wextra -Wpedantic)
endif()
