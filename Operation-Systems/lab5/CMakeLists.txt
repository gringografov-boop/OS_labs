cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(lab5_variant13 LANGUAGES CXX)
endif()
if(NOT TARGET os)
    add_subdirectory(../common/code/os ${CMAKE_CURRENT_BINARY_DIR}/os)
endif()
foreach(i 1 2)
    add_library(lab5_impl${i} SHARED code/impl${i}.cpp)
    target_compile_definitions(lab5_impl${i} PRIVATE LAB5_EXPORT)
endforeach()
add_executable(lab5_static code/static.cpp)
target_link_libraries(lab5_static PRIVATE lab5_impl1)
add_executable(lab5_dynamic code/dynamic.cpp)
target_link_libraries(lab5_dynamic PRIVATE os ${CMAKE_DL_LIBS})
add_dependencies(lab5_dynamic lab5_impl1 lab5_impl2)
foreach(t lab5_impl1 lab5_impl2 lab5_static lab5_dynamic)
    target_compile_features(${t} PRIVATE cxx_std_17)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/code/include)
    set_target_properties(${t} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/lab5
        LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/lab5
        ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib/lab5)
    foreach(c DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(${t} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY_${c} ${CMAKE_BINARY_DIR}/bin/lab5
            LIBRARY_OUTPUT_DIRECTORY_${c} ${CMAKE_BINARY_DIR}/bin/lab5)
    endforeach()
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
        target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    else()
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(UNIX)
    set_target_properties(lab5_static PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
endif()
